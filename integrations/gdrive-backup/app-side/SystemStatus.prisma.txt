// ── App-side wiring for the Google Drive backup / Apps Script integration ──
// Add the block below to prisma/schema.prisma (e.g. after the Notification
// model), then run `prisma migrate dev --name add_system_status`.
//
// Why it exists: the Apps Script keepalive + backup jobs write their results
// here (last backup time/file/checksum, keepalive health). The admin dashboard
// reads it via the BackupStatusPanel component, so the app, the script, and
// Drive's backup-manifest.json all share one source of truth.
// No new environment variables are required on the app side.

model SystemStatus {
  key       String   @id
  value     String
  updatedAt DateTime @updatedAt
}
